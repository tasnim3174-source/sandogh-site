Sub CalculateWaterBill() Dim ws As Worksheet Dim newWs As Worksheet Dim i As Integer ' 1. تعریف متغیرها بر اساس اطلاعات شما Dim totalBill As Double: totalBill = 38642000 ' جمع کل مبلغ قبض Dim totalWater As Double: totalWater = 20022288 ' آب بها Dim totalSewage As Double: totalSewage = 14386258 ' فاضلاب Dim totalTax As Double: totalTax = 2494375 ' مالیات و عوارض Dim totalLegal As Double: totalLegal = 739200 ' تکالیف قانونی Dim mainMeter As Double: mainMeter = 204 ' کارکرد کنتور اصلی Dim pattern As Double: pattern = 13 ' الگوی مصرف ' 2. ایجاد یک شیت جدید برای خروجی On Error Resume Next Application.DisplayAlerts = False Sheets("تسویه حساب").Delete Application.DisplayAlerts = True On Error GoTo 0 Set newWs = Sheets.Add(After:=Sheets(Sheets.Count)) newWs.Name = "تسویه حساب" ' 3. محاسبات پایه Dim totalVariable As Double totalVariable = totalWater + totalSewage + totalTax ' کل هزینه متغیر Dim basePrice As Double basePrice = totalVariable / mainMeter ' قیمت پایه هر مترمکعب ' مصرف واحدها Dim units(1 To 9) As Double units(1) = 16: units(2) = 10: units(3) = 26: units(4) = 29: units(5) = 21 units(6) = 10: units(7) = 19: units(8) = 22: units(9) = 6 ' محاسبه اختلاف (مشاعات) Dim sumUnits As Double: sumUnits = 0 For i = 1 To 9 sumUnits = sumUnits + units(i) Next i Dim sharedWater As Double: sharedWater = mainMeter - sumUnits ' 45 متر مکعب Dim sharedCost As Double: sharedCost = sharedWater * basePrice ' هزینه ریالی مشاعات Dim sharedPerUnit As Double: sharedPerUnit = sharedCost / 9 ' سهم هر واحد Dim legalPerUnit As Double: legalPerUnit = totalLegal / 9 ' سهم تکالیف قانونی ' 4. محاسبه واحدهای معادل (با ضریب جریمه 2 برابر برای مازاد الگو) Dim equivUnits(1 To 9) As Double Dim sumEquiv As Double: sumEquiv = 0 For i = 1 To 9 If units(i) <= pattern Then equivUnits(i) = units(i) Else equivUnits(i) = pattern + ((units(i) - pattern) * 2) End If sumEquiv = sumEquiv + equivUnits(i) Next i Dim equivPrice As Double ' مبلغ کل متغیر منهای هزینه مشاعات، تقسیم بر مجموع واحدهای معادل equivPrice = (totalVariable - sharedCost) / sumEquiv ' 5. ساخت جدول در شیت جدید With newWs ' تنظیم عنوان‌ها .Cells(1, 1).Value = "ردیف" .Cells(1, 2).Value = "واحد" .Cells(1, 3).Value = "مصرف کنتور" .Cells(1, 4).Value = "مازاد الگو" .Cells(1, 5).Value = "واحد معادل" .Cells(1, 6).Value = "هزینه آب اختصاصی" .Cells(1, 7).Value = "سهم مشاعات" .Cells(1, 8).Value = "تکالیف قانونی" .Cells(1, 9).Value = "مبلغ کل (ریال)" ' استایل هدر .Range("A1:I1").Font.Bold = True .Range("A1:I1").Interior.Color = RGB(200, 220, 255) .Range("A1:I1").HorizontalAlignment = xlCenter Dim totalFinal As Double: totalFinal = 0 ' پر کردن داده‌ها For i = 1 To 9 Dim excess As Double If units(i) > pattern Then excess = units(i) - pattern Else excess = 0 Dim waterCost As Double waterCost = equivUnits(i) * equivPrice Dim finalCost As Double finalCost = waterCost + sharedPerUnit + legalPerUnit .Cells(i + 1, 1).Value = i .Cells(i + 1, 2).Value = "واحد " & i .Cells(i + 1, 3).Value = units(i) .Cells(i + 1, 4).Value = excess .Cells(i + 1, 5).Value = equivUnits(i) .Cells(i + 1, 6).Value = Round(waterCost, 0) .Cells(i + 1, 7).Value = Round(sharedPerUnit, 0) .Cells(i + 1, 8).Value = Round(legalPerUnit, 0) .Cells(i + 1, 9).Value = Round(finalCost, 0) totalFinal = totalFinal + Round(finalCost, 0) Next i ' جمع کل .Cells(11, 2).Value = "جمع کل" .Cells(11, 3).Value = sumUnits .Cells(11, 5).Value = sumEquiv .Range("B11:I11").Font.Bold = True .Range("B11:I11").Interior.Color = RGB(230, 230, 230) .Cells(11, 9).Value = totalFinal ' قالب‌بندی اعداد و ستون‌ها .Range("C2:E10").NumberFormat = "0" .Range("F2:I11").NumberFormat = "#,##0" .Columns("A:I").AutoFit End With MsgBox "محاسبات با موفقیت انجام شد و در شیت 'تسویه حساب' قرار گرفت.", vbInformation, "پایان محاسبات" End Sub
